// 2. 解析 SPS 和 PPS
                            // 注意：Android MediaCodec 输出的 config buffer 通常已经是 Annex-B 格式 (带 00 00 00 01 起始码)
                            // 或者可能是 AVCC 格式，具体取决于编码器配置。通常需要解析 NALU 类型。
                            // H.264: NALU Type 7 is SPS, Type 8 is PPS.

                            var sps: ByteArray? = null
                            var pps: ByteArray? = null

                            // 简单解析逻辑示例 (假设是 Annex-B 格式)
                            var offset = 0
                            while (offset < size) {
                                // 查找起始码 0x00000001 或 0x000001
                                if (offset + 3 < size && data[offset] == 0.toByte() && data[offset + 1] == 0.toByte() && data[offset + 2] == 0.toByte() && data[offset + 3] == 1.toByte()) {
                                    val nalStart = offset + 4
                                    var nextNalStart = size // 默认到末尾

                                    // 寻找下一个起始码
                                    for (i in nalStart until size - 3) {
                                        if (data[i] == 0.toByte() && data[i + 1] == 0.toByte() && data[i + 2] == 0.toByte() && data[i + 3] == 1.toByte()) {
                                            nextNalStart = i
                                            break
                                        }
                                    }

                                    // 获取 NALU 类型 (去掉起始码后的第一个字节的低5位)
                                    val nalType = data[nalStart].toInt() and 0x1F

                                    if (nalType == 7) { // SPS
                                        sps = data.copyOfRange(nalStart, nextNalStart)
                                    } else if (nalType == 8) { // PPS
                                        pps = data.copyOfRange(nalStart, nextNalStart)
                                    }

                                    offset = nextNalStart
                                } else {
                                    offset++
                                }
                            }

                            // 3. 保存或使用 SPS/PPS
                            if (sps != null && pps != null) {
                                logInfo("SPS size: ${sps.size}, PPS size: ${pps.size}")

                                // e.g., rtmpPublisher.sendConfig(sps, pps)
